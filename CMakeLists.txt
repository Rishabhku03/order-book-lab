cmake_minimum_required(VERSION 3.25)
project(lob_replay VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LOB_SANITIZE "Address and undefined behavior sanitizers" OFF)
option(LOB_COVERAGE "Clang source coverage" OFF)
option(LOB_FUZZ "Build libFuzzer targets" OFF)
option(LOB_BUILD_BENCHMARKS "Build microbenchmarks" ON)
option(LOB_WARNINGS_AS_ERRORS "Reject warnings in project code" ON)
include(cmake/Dependencies.cmake)
add_library(lob_core INTERFACE)
target_include_directories(lob_core INTERFACE include)
if(MSVC)
  target_compile_options(lob_core INTERFACE /W4 /permissive- /EHsc)
else()
  target_compile_options(lob_core INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(LOB_WARNINGS_AS_ERRORS)
  if(MSVC)
    target_compile_options(lob_core INTERFACE /WX)
  else()
    target_compile_options(lob_core INTERFACE -Werror)
  endif()
endif()
if(LOB_SANITIZE)
  target_compile_options(lob_core INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(lob_core INTERFACE -fsanitize=address,undefined)
  if(WIN32)
    # MSVC STL container annotations must agree in every object linked into a test.
    target_compile_options(Catch2 PRIVATE -fsanitize=address,undefined)
    target_compile_options(Catch2WithMain PRIVATE -fsanitize=address,undefined)
  endif()
endif()
if(LOB_COVERAGE)
  target_compile_options(lob_core INTERFACE -fprofile-instr-generate -fcoverage-mapping)
  target_link_options(lob_core INTERFACE -fprofile-instr-generate)
endif()
execute_process(COMMAND git rev-parse --short=12 HEAD WORKING_DIRECTORY ${CMAKE_SOURCE_DIR} OUTPUT_VARIABLE LOB_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT LOB_COMMIT)
  set(LOB_COMMIT "uncommitted")
endif()
add_executable(lob src/main.cpp)
target_link_libraries(lob PRIVATE lob_core nlohmann_json::nlohmann_json)
target_include_directories(lob SYSTEM PRIVATE ${picosha_SOURCE_DIR})
target_compile_definitions(lob PRIVATE LOB_COMMIT="${LOB_COMMIT}" LOB_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
if(WIN32)
  target_link_libraries(lob PRIVATE psapi)
endif()
include(CTest)
add_executable(lob_tests tests/test_core.cpp)
target_link_libraries(lob_tests PRIVATE lob_core Catch2::Catch2WithMain)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang" AND CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 23)
  # Use Catch2's standard line-based names instead of the __COUNTER__ extension.
  target_compile_definitions(lob_tests PRIVATE CATCH_CONFIG_NO_COUNTER)
endif()
list(APPEND CMAKE_MODULE_PATH ${catch2_SOURCE_DIR}/extras)
include(Catch)
catch_discover_tests(lob_tests)
if(LOB_BUILD_BENCHMARKS)
  add_executable(lob_microbench bench/micro.cpp)
  target_link_libraries(lob_microbench PRIVATE lob_core benchmark::benchmark_main)
endif()
if(LOB_FUZZ)
  foreach(target parser state)
    add_executable(fuzz_${target} fuzz/${target}.cpp)
    target_link_libraries(fuzz_${target} PRIVATE lob_core)
    target_compile_options(fuzz_${target} PRIVATE -fsanitize=fuzzer)
    target_link_options(fuzz_${target} PRIVATE -fsanitize=fuzzer)
    if(WIN32)
      # Match the prebuilt LLVM libFuzzer's MSVC STL annotation ABI.
      # Normal sanitizer tests retain container annotations.
      target_compile_definitions(fuzz_${target} PRIVATE _DISABLE_STRING_ANNOTATION _DISABLE_VECTOR_ANNOTATION)
    endif()
  endforeach()
endif()
